Новые возможности

Переменные let и const

В ES-2015 предусмотрены новые способы объявления переменных: через let и const


    let x;

    const y = 10;

let

  • Блочная область видимости вместо лексической у var
  • Нельзя повторно объявлять в этом же блоке
  • Отсутствует hoisting (let и const не существуют до своего объявления)
  • При использовании в цикле, для каждой итерации создаётся своя переменная

    let x = 5;

    {
        let x = 10;
        console.log(x); // => 10
    }

    console.log(x); // => 5

const

  • Также имеет блочную область видимости
  • В const одновременно с объявлением переменной должно быть присвоено значение
  • Рекомендуется использовать UPPER_CASE
  • Разрешается переопределять свойства объекта

    const obj = {x: 5, y: 10};

    obj.x = 8;

    obj = []; // Uncaught TypeError: Assignment to constant variable.

Template Strings

  • Используются обратные кавычки ` (backtick)
  • ${ ... } используется для вычисляемого значения / переменной / выражения
  • Могут быть многострочными

    // было
    $('[data-param="' + x + '-' + y + '"]').on

    // стало
    $(`[data-param="${x}-${y}"]`).on

Многострочность


    let title = `Hello, world!`;
    let body = `Lorem ipsum dolor sit amet, consectetur adipisicing elit.`;

    let article = `<article class="article">
            <div class="article__title">${title.toUpperCase()}</div>
            <div class="article__body">${body}</div>
        </article>`;

Функции шаблонизации

Можно использовать свою функцию шаблонизации для строк. Название этой функции ставится перед первой обратной кавычкой:


    let str = func`моя строка`;

Пример


    function toUSD(strings, ...vals) {
        return strings.reduce((previousValue, currentItem, index) => {
            return previousValue + Math.floor(vals[index - 1] / 27) + currentItem;
        });
    }

    let str = toUSD`Какая-то строка, ${6200} продолжение строки.`;

    console.log(str);
    // "Какая-то строка, 229 продолжение строки."

Стрелочные функции

В ES-2015 появился новый синтаксис для задания функций через «стрелку» =>


    // было
    let pow = function(x) {
        return x * x;
    };

    // стало, краткая запись
    let pow = x => x * x;

    // блочная запись
    let pow = (x) => {
        return x * x;
    };

что нужно знать

  • Неявный return при отсутствии {}
  • Не имеют собственного контекста, this — тот же что у родительской области
  • Стрелочные функции не могут быть использованы как конструктор

Неявный return


    let arr = [1,2,3,4,5,6];

    let filterArr = arr.filter(x => x % 2 !== 0);

    // filterArr = [1,3,5]

Отсутствие контекста


    function Person() {
        this.name = `John`;

        setInterval(() => {
            console.log(this.name); // 'John'
        }, 1000);
    }

Параметры по умолчанию

В ES2015 появилась возможность указывать параметры по умолчанию


    let hello = (user=`Guest`) => {
        return `Hello, ${user}!`;
    };

    hello(`world`); // "Hello, world!"

    hello(); // "Hello, Guest!"

операторы spread & rest

  • Имеют одинаковый синтаксис "..."
  • Различная интерпритация, в зависимости от контекста применения

spread

Позволяет разложить массив на составляющие


    //пример 1
    let arr = [2, 3, 5];

    let func = (a,b,c) => {
        //...
    };

    func(...arr);

    // пример 2
    let data = [...arr, 7, 11, 13];

rest

Позволяет объединить отдельные значения в массив


    let func = (first, ...rest) => {
        console.log(first); // 2
        console.log(rest); // [3,5,7]
    }

    func(2,3,5,7);

Деструктуризация

Деструктуризация (destructuring assignment) – новый синтаксис присваивания, при котором можно присвоить массив или объект сразу нескольким переменным, разбив его на части.

Деструктуризация массива


    let array = [1, 2, 3, 4];

    let [first, , third] = array;

    console.log(first, third); // 1 3

в связке с оператором rest


    let array = [1, 2, 3, 4];

    let [first, ...rest] = array;

    console.log(first, rest); // 1 [2, 3, 4]

обмен значениями


    let x = 5;
    let y = 10;

    [x, y] = [y, x];

    console.log(x, y); // 10 5

Значения по умолчанию


    let array = [];
    let [userName=`Guest`] = array;

Деструктуризация объекта


    let time = {
        year: `2016`,
        month: `12`,
        day: `26`,
        hour: `12`,
        minute: `30`
    };

    let {year, day} = time;

присвоение свойства объекта в переменную с другим именем


    let user = {name: `John`, age: 25};
    let {name: userName} = user;

    console.log(userName); // John

Вложенные деструктуризации


    let users = [
        {name: `John`},
        {name: `Bill`}
    ];

    let [, {name: secondUser}] = users;

    console.log(secondUser); // Bill

Деструктуризация параметров функции


    let user = {name: `John`, age: 25};

    let func = ({name=`Guest`, age}) => {
        console.log(`${name}, ${age}`);
    };

    func(user); // John, 25

Object Literals

  • Сокращенная запись для значения свойства
  • Вычисляемые имена свойства
  • Сокращенная запись для методов объекта

Сокращенная запись свойств объекта

Когда имя свойства совпадает с его значением, то последнее можно опустить, в ES6 оно подставляется неявно.


    let name = `John`;
    let age = 25;

    // старая запись
    let user = {
        name: name,
        age: age
    };

    // сокращенная запись
    user = {name, age};

Вычисляемые имена свойства

Динамические имена свойств позволяют писать выражения, обёрнутые в квадратные скобки. Результат выражения станет именем свойства.


    let username = `name`;

    let user = {
        [username]: `John`,
        age: 25
    };

    console.log(user.name); // John

Сокращенная запись методов

Объявление методов объекта больше не требует ключевого слова function


    let obj = {
        count: 4,

        getCount() {
            return this.count;
        }
    };

    obj.getCount(); // 4

Классы

В ES6 появился новый синтаксис для классов. Класс в ES6 не представляет собой новую объектно-ориентированную модель наследования, а просто новый синтаксис для работы с прототипами и функциями-конструкторами.

Пример


    class User {
        constructor(name) {
            this.name = name;
        }

        getName() {
            return this.name;
        }
    };

    let user = new User(`John`);

    user.getName(); // John

constructor

Функция constructor запускается при создании экземпляра объекта, остальные методы записываются в prototype

Геттеры, сеттеры

В классах, как и в обычных объектах, можно объявлять геттеры и сеттеры через get/set, а также использовать […] для свойств с вычисляемыми именами:

пример


    class User {
        constructor(name) {
            this.name = name;
        }

        get userAge() {
            return this.age;
        }

        set userAge(age) {
            this.age = age;
        }
    };

    let user = new User(`John`);

    user.userAge = 25;

    console.log(user.userAge); // 25

Статические методы

Для того, чтобы объявить статический метод класса, необходимо использовать слово static.

Статические методы класса – это методы непосредственно самого класса.

Статические методы вызываются через имя класса. Вызывать статические методы через имя объекта запрещено.

пример


    class User {
        constructor(name) {
            this.name = name
        }

        static isStatic() {
            return true;
        }
    }

    let user = new User(`John`);

    user.isStatic(); // user.isStatic is not a function

    User.isStatic(); // true

Наследование

В ES6 ключевое слово extends позволяет классу-потомку наследовать от родительского класса. Важно отметить, что конструктор класса-потомка должен вызывать super().

Также, в классе-потомке можно вызвать метод родительского класса с помощью super.имяМетодаРодителя().

пример


    class User {
        constructor(name) {
            this.name = name;
        }

        getName() {
            return this.name;
        }
    };

    class Admin extends User {
        constructor(name) {
            super(name);

            this.role = `admin`;
        }
    };

    let user = new Admin(`John`);

    user.getName(); // John

Symbol

Symbol это уникальный и неизменяемый тип данных, представленный в ES2015. Целью Symbol является создание уникального идентификатора, к которому нельзя получить доступ.

Создание


    let symbol = Symbol();

    console.log(typeof symbol); // symbol

У функции Symbol есть необязательный аргумент «имя». Его можно использовать для описания символа.


    let symbol_1 = Symbol(`name`);
    let symbol_2 = Symbol(`name`);

    console.log(symbol_1 === symbol_2); // false

Глобальные символы

Существует «глобальный реестр» символов, который позволяет, при необходимости, иметь общие «глобальные» символы, которые можно получить из реестра по имени.

Метод Symbol.for(key) ищет ключ в «глобальном реестре». Если символ с этим ключом есть, то этот символ и вернется. Если символ с таким ключом не найден, то он создаётся.

Создание символа в «глобальном реестре»


    let name = Symbol.for(`name`);

    console.log(Symbol.for(`name`) == name); // true

Получение имени символа


    let name = Symbol.for(`symbolName`);

    Symbol.keyFor(name); // "symbolName"

Использование символов

Символы можно использовать в качестве имён для свойств объекта


    let user = {
        login: `wezom`,
        [Symbol.for(`password`)]: `wezom`
    };

    console.log(user[Symbol.for(`password`)]); // "wezom"

Итераторы

В современный JavaScript добавлена новая концепция «итерируемых» (iterable) объектов.

Итерируемые или, иными словами, «перебираемые» объекты – это те, содержимое которых можно перебрать в цикле.

Например, перебираемым объектом является массив. Но не только он. В браузере существует множество объектов, которые не являются массивами, но содержимое которых можно перебрать (к примеру, список DOM-узлов).

Symbol.iterator

В ES2015 у всех «итерируемых» объектов есть метод Symbol.iterator, который определяет итератор для объекта по-умолчанию.


    let arr = [1,2,3,4,5];

    console.log(typeof arr[Symbol.iterator]); // "function"

У итератора есть метод next(), который возвращает следующий элемент в последовательности. Этот метод возвращает объект с двумя свойствами: done (окончен ли перебор) и value (значение).


    let arr = [2,3,5];

    let itr = arr[Symbol.iterator]();

    itr.next(); // { value: 2, done: false }
    itr.next(); // { value: 3, done: false }
    itr.next(); // { value: 5, done: false }
    itr.next(); // { value: undefined, done: true }

for..of

Для перебора «итерируемых» объектов добавлен новый синтаксис цикла: for..of


    for (let char of `Hello`) {
        console.log(char); // Выведет по одной букве: H, e, l, l, o
    }

    let arr = [2,3,5];

    for (let item of arr) {
        console.log(item); // 2, 3, 5
    }

Объекты Array, String, Map, Set, arguments, NodeList — итерируемые в ES6 по умолчанию

Также можно написать собственный итератор через определение obj[Symbol.iterator]() с описанием объекта.

Генераторы

Генераторы – новый вид функций в современном JavaScript. Они отличаются от обычных тем, что могут приостанавливать своё выполнение, возвращать промежуточный результат и далее возобновлять его позже, в произвольный момент времени.

Создание генератора

Для объявления генератора используется новая синтаксическая конструкция: function*


    function* generateAnything() {
        // ...
    }

Функция-генератор возвращает итерируемый объект при своём вызове

yield

Внутри функции-генератора доступно ключевое слово yield. При вызове функции-генератора она выполняет код до ближайшего ключевого слова yield. По достижении yield выполнение приостанавливается, а значение – возвращается

Пример


    function* range(start, stop) {
        for (let i = start; i <= stop; i++) {
            yield i;
        }
    }

    let counter = range(5, 7);

    counter.next(); // {"done": false, "value": 5}
    counter.next(); // {"done": false, "value": 6}
    counter.next(); // {"done": false, "value": 7}
    counter.next(); // {"done": true, "value": undefined}

Перебор через for..of


    function* generateNumber() {
        yield 1;
        yield 2;
        yield 3;
    }

    let numbers = generateNumber();

    for(let value of numbers) {
        console.log(value); // 1, 2, 3
    }

spread


    console.log([...numbers]); // [1,2,3]

Делегирование генератора


    function* generateNumber() {
        yield 1;
        yield 2;
        yield 3;
    }

    function* generate() {
        yield `start`;
        yield* generateNumber();
        yield `finish`;
    }

    let result = generate();

    result.next(); // {"done": false, "value": "start"}
    result.next(); // {"done": false, "value": 1}
    result.next(); // {"done": false, "value": 2}
    result.next(); // {"done": false, "value": 3}
    result.next(); // {"done": false, "value": "finish"}

Передача значения извне в генератор

yield не только возвращает результат наружу, но и может передавать значение извне в генератор


    function* generate() {
        let input = yield `start`;

        console.log(input); // 1

        input = yield;

        console.log(input); // 2

        yield;
    }

    let result = generate();

    result.next(0); // {"done": false,"value": "start"}
    result.next(1); // {"done": false,"value": undefined}
    result.next(2); // {"done": false,"value": undefined}

Promises

В ES6 появилась встроенная поддержка промисов. Промис это объект, который ждёт выполнения асинхронной операции, после которого (т.е. после выполнения) промис принимает одно из двух состояний: fulfilled (resolved, успешное выполнение) или rejected (выполнено с ошибкой).

Стандартным способом создания промиса является конструктор new Promise(), который принимает обработчик с двумя функциями как параметрами. Первый обработчик (обычно именуемый resolve) представляет собой функцию для вызова вместе с будущим значением, когда оно будет готово; второй обработчик (обычно именуемый reject) является функцией, которая вызывается для отказа от выполнения промиса

Создание промиса


    let promise = new Promise(function(resolve, reject) {
        if (/* условие */) {
            resolve(/* значение */);  // fulfilled successfully
        } else {
            reject(/* reason */);  // rejected
        }
    });

Каждый промис обладает методом then, в котором есть два callback-a. Первый callback вызывается, если промис успешно выполнен (resolved), тогда как второй callback вызывается, если промис выполнен с ошибкой (rejected).


    let promise = new Promise(function(resolve, reject) {
        resolve(`success`);
    });

    promise
        .then(
            response => {
                console.log(response); // "success"
            }
        );

Использование

В современном JavaScript промисы используются для организации асинхронного кода.


    function httpGet(url) {
        return new Promise(function(resolve, reject) {

            var xhr = new XMLHttpRequest();
            xhr.open('GET', url, true);

            xhr.onload = function() {
                if (this.status == 200) {
                    resolve(this.response);
                } else {
                    var error = new Error(this.statusText);
                    error.code = this.status;
                    reject(error);
                }
            };

            xhr.onerror = function() {
                reject(new Error("Network Error"));
            };

            xhr.send();
        });
    }

    httpGet("/article/promise/user.json")
        .then(
            response => {
                console.log(`Success: ${response}`);
            },
            error => {
                console.log(`Error: ${error}`);
            }
        );

Цепочки промисов

Возможность строить асинхронные цепочки из промисов – пожалуй, основная причина, из-за которой существуют и активно используются промисы.

При последовательных вызовах .then…then…then, в каждый следующий then переходит результат от предыдущего.


    function getDate() {
        return new Promise((resolve, reject) => {
            setTimeout(() => {
                resolve(new Date());
            }, 2000);
        });
    }

    getDate()
        .then(date => {
            return `Date: ${date}`;
        })
        .then(result => {
            console.log(result); // "Date: Thu Dec 22 2016 20:43:38..."
            return getDate().then((date) => date);
        })
        .then(result => {
            console.log(`Date: ${result}`); // "Date: Thu Dec 22 2016 20:43:40..."
        });

Параллельное выполнение

Вызов метода Promise.all(iterable) получает массив (или другой итерируемый объект) промисов и возвращает промис, который ждёт, пока все переданные промисы завершатся, и переходит в состояние «выполнено» с массивом их результатов.

метод Promise.all


    Promise.all([
        httpGet('/article/promise/user.json'),
        httpGet('/article/promise/guest.json')
    ]).then(results => {
        console.log(results); // [...]
    });

метод Promise.race

Вызов Promise.race, как и Promise.all, получает итерируемый объект с промисами, которые нужно выполнить, и возвращает новый промис.

Но, в отличие от Promise.all, результатом будет только первый успешно выполнившийся промис из списка. Остальные игнорируются.

Map & WeakMap

В ES-2015 появились новые структуры данных: Set, Map, WeakSet и WeakMap.

Map

Map – коллекция для хранения записей вида ключ:значение.

В отличие от объектов, в которых ключами могут быть только строки, в Map ключом может быть что угодно: функции, объекты, даты...

Создание Map


    let map = new Map();

    map.set(`name`, `John`);
    map.set(false, 42);
    map.set({value: 10}, [2, 3, 5, 7]);
    map.set(new Date(), `Hello, world!`);
    map.set(document.querySelector(`body`), `DOM Element`);

    map.get(false); // 42

Методы

  • map.set(key, value) – запись значения
  • map.get(key) – получения значения по ключу
  • map.size – общее количество записей в map
  • map.delete(key) – удаляет запись с ключом key
  • map.clear() – удаляет все записи, очищает map
  • map.has(key) – возвращает true, если ключ есть, иначе false

Перебор

Для итерации по map используется один из трёх методов:

  • map.keys() – возвращает итерируемый объект для ключей
  • map.values() – возвращает итерируемый объект для значений
  • map.entries() – возвращает итерируемый объект для записей [ключ, значение], он используется по умолчанию в for..of

Map является итерируемым объектом, значит его можно перебрать с помощью for..of


    let users = new Map();

    users
        .set(`1`, `John`)
        .set(`2`, `Bill`)
        .set(`3`, `Steve`);

    for(let item of users.keys()) {
        console.log(item); // "1", "2", "3"
    }

WeakMap

Можно рассматривать объект WeakMap как подмножество Map. У WeakMap есть несколько ограничений, отсутствующих в Map. Самое главное ограничение WeakMap в том, что в отличие от Map он не итерируемый, а значит нет методов .entries(), .keys(), values(), .forEach() и .clear()

Ключи WeakMap должны быть ссылочных типов. В виде ключей нельзя использовать такие типы, как символы, числа или строки.


    let map = new WeakMap();

    let obj = {};
    let header = document.querySelector(`.header`);

    map.set(obj, `hello, world!`);

    map.set(header, {
        show(){ /* ... */ },
        hide(){ /* ... */ }
    });

В WeakMap ключи обладают неустойчивыми связями, что позволяет не мешать сборщику мусора удалять элементы WeakMap. Это означает, что можно не беспокоиться об утечках памяти.

Записи WeakMap с ключом key, который является единственной ссылкой на указанную переменную — подлежат сборке мусора

Set & WeakSet

Объекты Set это коллекции уникальных значений. Дублированные значения игнорируются, т.к. коллекция должна содержать только уникальные значения. Значения могут быть примитивами или ссылками на объекты.

У объектов Set нет ключей, есть только значения


    var set = new Set([1, 2, 3, 4, 4]);

    console.log([...set]); // [1, 2, 3, 4]

Методы

  • set.add(item) – добавляет в коллекцию item
  • set.delete(item) – удаляет item из коллекции, возвращает true, если он там был, иначе false
  • set.has(item) – возвращает true, если item есть в коллекции, иначе false
  • set.clear() – очищает set

Объект Set является итерируемым. Можно перебирать Set в цикле с помощью for...of. Перебор происходит в том же порядке, что и вставка.


    let users = new Set();

    users
        .add(`John`)
        .add(`Bill`)
        .add(`John`)
        .add(`Steve`);

    for(let item of users) {
        console.log(item); // "John", "Bill", "Steve"
    }

WeakSet

Аналогично WeakMap, объект WeakSet позволяет хранить объекты с неустойчивыми связями в коллекции.

Объект WeakSet не итерируемый.

Это значит, что нельзя перебирать WeakSet, его значения должны быть ссылками на уникальный объект. Если на найденное в WeakSet value ни на что не ссылается, объект подлежит сборке мусора.

Пример


    let set = new WeakSet();
    let date = new Date();

    set.add(date);
    set.has(date);  // true

ES2015 Modules

По сути, модуль — это инструкция (statement), которая вызывается неявно — посредством создания файла и выполнения его с помощью интерпретатора ES (прямо, при «запуске» файла программистом, или косвенно, в результате импорта другим модулем). В ES6 есть чёткое соотношение: один файл — один модуль. Каждый модуль имеет отдельную область видимости (Lexical environment) — т. е. все объявления переменных, функций и классов не будут доступны за пределами модуля (файла), если не экспортированы явно.

export

Оператор export позволяет экспортировать сущности модуля, чтобы они были доступны из других модулей.


    export let obj = {};

    export class User {
        // ...
    };

    export const PI = 3.14159265359;

Экспорт объявленной сущности


    function func(){ /* ... */ };

    const PI = 3.14159265359;

    export {func, PI};

export as


    class User {
        // ...
    };

    export {User as UserAPI};

export default

В случае когда модуль экспортирует только одно значение, удобно определить это значение как экспортируемое по умолчанию:


    export default class {
        constructor(id, name) {
            // ...
        }
    };

import

Другие модули могут подключать экспортированные значения при помощи ключевого слова import.


    import { User, PI } from './api/math';

Импортировать можно под другим именем при помощи «as»:


    import {list as users} from './users';

Можно импортировать все значения в виде объекта при помощи import * as obj from "module"


    import * as mathAPI from './api/math';

Свойство экспортированное по умолчанию при импорте присваивается без фигурных скобок


    import User from './api/user';

Экспортированное по умолчанию свойство при импорте можно назвать как угодно.

Реэкспорт

Повторный экспорт модулем свойства, которое он импортирует из другого модуля


    export { something } from './another';

    export * from './another';

Proxy

Прокси (proxy) – особый объект, смысл которого – перехватывать обращения к другому объекту и, при необходимости, модифицировать их.

Синтаксис


    let proxy = new Proxy(target, handler);

target – объект, обращения к которому надо перехватывать

handler – объект с «ловушками»: функциями-перехватчиками для операций к target

Почти любая операция может быть перехвачена и обработана прокси до или даже вместо доступа к объекту target, например: чтение и запись свойств, получение списка свойств, вызов функции (если target – функция)

методы обработчика

  • getPrototypeOf – перехватывает обращение к методу getPrototypeOf.
  • setPrototypeOf – перехватывает обращение к методу setPrototypeOf.
  • isExtensible – перехватывает обращение к методу isExtensible.
  • preventExtensions – перехватывает обращение к методу preventExtensions.
  • getOwnPropertyDescriptor – перехватывает обращение к методу getOwnPropertyDescriptor.
  • defineProperty – перехватывает обращение к методу defineProperty.
  • has – перехватывает проверку существования свойства, которая используется в операторе in и в некоторых других методах встроенных объектов.
  • get – перехватывает чтение свойства.
  • set – перехватывает запись свойства.
  • deleteProperty – перехватывает удаление свойства оператором delete.
  • enumerate – срабатывает при вызове for..in или for..of, возвращает итератор для свойств объекта.
  • ownKeys – перехватывает обращения к методу getOwnPropertyNames.
  • apply – перехватывает вызовы target().
  • construct – перехватывает вызовы new target().

get

Срабатывает при чтении свойства из прокси.


    get(target, property, receiver);
  • target – целевой объект, тот же который был передан первым аргументом в new Proxy.
  • property – имя свойства.
  • receiver – объект, к которому было применено присваивание. Обычно сам прокси, либо прототипно наследующий от него. Этот аргумент используется редко.

get


    let user = {
        name: `john`
    };

    let proxy = new Proxy(user, {
        get(target, prop) {
            console.log(`Чтение ${prop}`); // "Чтение name"
            return target[prop].toUpperCase();
        }
    });

    console.log(proxy.name); // "JOHN"

set

Срабатывает при записи свойства в прокси.


    set(target, property, value, receiver);
  • target – целевой объект, тот же который был передан первым аргументом в new Proxy.
  • property – имя свойства.
  • value – значение свойства.
  • receiver – объект, к которому было применено присваивание, обычно сам прокси, либо прототипно наследующий от него.

set

Метод set должен вернуть true, если присваивание успешно обработано и false в случае ошибки 


    let user = {};

    let proxy = new Proxy(user, {
        set(target, prop, value) {
            console.log(`Запись ${prop} ${value}`); // "Запись name John"
            target[prop] = value;
            return true;
        }
    });

    proxy.name = `John`;

has

Ловушка has срабатывает в операторе in и некоторых других случаях, когда проверяется наличие свойства.


    let user = {
        "name": "John"
    };

    let proxy = new Proxy(user, {
        has(target, phrase) {
            console.log(phrase); // "name"

            if(phrase in target) {
                return true;
            } else {
                return false;
            }
        }
    });

    console.log("name" in proxy); // true

apply

Прокси умеет работать не только с обычными объектами, но и с функциями. Если аргумент target прокси – функция, то становится доступна ловушка apply для её вызова.


    let obj = {ratio: 42};

    function calcSomething(num) {
        return num * this.ratio;
    }

    let proxy = new Proxy(calcSomething, {
        apply: function(target, context, argumentsList) {
            return target.apply(context, argumentsList);
        }
    });

    console.log( proxy.apply(obj, [10]) ); // 420

construct

Ловушка construct перехватывает вызовы при помощи new. Она получает исходный объект target и список аргументов argumentsList.


    function User(name, surname) {
        this.name = name;
        this.surname = surname;
    }

    let UserProxy = new Proxy(User, {
        construct: function(target, argumentsList) {
            return new target(...argumentsList);
        }
    });

    let user = new UserProxy(`John`, `Doe`);

    console.log(user.name); // "John"

Reflect API

Reflect - новый статический встроенный объект в ES2015, который позволяет создавать методы, перехватывающие операции JavaScript. При этом методы называются так же, как и методы proxy handler`ов. Reflect - это не функциональный, а простой объект, он не является конструктором.

Пример


    function User(name, surname) {
        this.name = name;
        this.surname = surname;
    }

    let ref = Reflect.construct(User, [`John`, `Doe`]);

    // let ref = new User(`John`, `Doe`);

    console.log(ref.name); // "John"

Object.assign

Метод Object.assign() используется для копирования значений всех собственных перечисляемых свойств из одного или более исходных объектов в целевой объект. После копирования он возвращает целевой объект.


    let object1 = {x: 1, y: 3};
    let object2 = {y: 5, z: 7};

    let object = Object.assign({}, object1, object2);

    // аналогично записи
    let object = $.extend({}, object1, object2);

    console.log(object); // {x: 1, y: 5, z: 7};
